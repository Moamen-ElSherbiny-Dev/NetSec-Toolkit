# Create a Python virtual environment named .venv
py -m venv .venv

# Activate the virtual environment using PowerShell
.\.venv\Scripts\Activate.ps1

# Activate the virtual environment using Command Prompt
.venv\Scripts\activate.bat

# Upgrade pip to the latest available version
python -m pip install --upgrade pip

# Install PyInstaller for packaging the Python application into an executable
pip install pyinstaller

# Install all required Python dependencies listed in requirements_Libs.txt
pip install -r requirements_Libs.txt

# Extract the version information from the Windows Notepad executable
pyi-grab_version "C:\Windows\System32\notepad.exe"
